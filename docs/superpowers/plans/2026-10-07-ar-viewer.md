# AR Viewer Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 3D 작업자가 GLB/glTF 파일을 열어 3D로 돌려보며 검토하는 오프라인 데스크톱 앱(macOS + Windows)을 만든다.

**Architecture:** Electron 앱이다. Main 프로세스는 창, 메뉴, 파일 대화상자, 커스텀 `model://` 프로토콜(허용된 폴더의 로컬 파일만 제공)을 맡고 외부 네트워크 요청은 모두 차단한다. Sandbox된 preload는 `window.api`로 필요한 API만 노출한다. React 렌더러는 툴바, 빈 화면, `<model-viewer>`를 보여준다. 렌더링 설정은 modelviewer.dev/editor 기본값으로 고정하고, Draco/KTX2/Meshopt 디코더는 앱에 포함한다.

**Tech Stack:** Electron 44, electron-vite 5 (Vite 7), React 19 + TypeScript 5.9, @google/model-viewer 4.3 (three 0.183), Vitest 4, Playwright 1.63 (Electron), electron-builder 26

**Spec:** `docs/superpowers/specs/2026-10-07-ar-viewer-design.md`

### Task 3: 경로 ↔ `model://` URL 변환

**Files:**
- Create: `src/main/model-url.ts`
- Test: `test/unit/model-url.test.ts`

URL 형식:
- macOS: `/Users/a b/x.glb` → `model://file/Users/a%20b/x.glb`
- Windows 드라이브: `C:\a\x.glb` → `model://file/C%3A/a/x.glb`
- Windows UNC(NAS 공유 폴더): `\\nas\share\x.glb` → `model://unc/nas/share/x.glb`

경로 조각마다 `encodeURIComponent`를 적용한다. 그래서 glTF가 `tex.png`를 상대 경로로 요청하면 브라우저가 `model://file/Users/a%20b/tex.png`로 알아서 해석한다.

describe('toModelUrl', () => {

    expect(toModelUrl('/Users/a b/의자 v3.glb', posix)).toBe(
      'model://file/Users/a%20b/%EC%9D%98%EC%9E%90%20v3.glb'

  it('encodes windows drive paths', () => {
    expect(toModelUrl('C:\\Work\\chair.glb', win32)).toBe('model://file/C%3A/Work/chair.glb')
  })

  it('uses the unc host for windows network shares', () => {
    expect(toModelUrl('\\\\nas\\share\\chair.glb', win32)).toBe('model://unc/nas/share/chair.glb')
  })

describe('fromModelUrl', () => {
  it('round-trips posix paths', () => {
    const p = '/Users/a b/의자 v3.glb'
    expect(fromModelUrl(toModelUrl(p, posix), posix)).toBe(p)
  })

  it('round-trips windows drive paths', () => {
    const p = 'C:\\Work\\의자.gltf'
    expect(fromModelUrl(toModelUrl(p, win32), win32)).toBe(p)
  })

  it('round-trips windows unc paths', () => {
    const p = '\\\\nas\\share\\chair.glb'
    expect(fromModelUrl(toModelUrl(p, win32), win32)).toBe(p)
  })

  it('resolves sibling resources relative to the model url', () => {
    const base = toModelUrl('/Users/a b/Duck.gltf', posix)
    expect(fromModelUrl(new URL('DuckCM.png', base).href, posix)).toBe('/Users/a b/DuckCM.png')

  it('collapses dot segments, including encoded ones', () => {
    expect(fromModelUrl('model://file/Users/a/../b/x.glb', posix)).toBe('/Users/b/x.glb')

    expect(fromModelUrl('https://example.com/x.glb', posix)).toBeNull()

  it('accepts files in the folder or its subfolders', () => {

    expect(isInside('/m', '/m', posix)).toBe(false)
    expect(isInside('/m', '/a.glb', posix)).toBe(false)
    expect(isInside('/m', '/m2/a.glb', posix)).toBe(false)
    expect(isInside('/m', '/m/../etc/passwd', posix)).toBe(false)
  })

  it('accepts names that merely start with two dots', () => {
    expect(isInside('/m', '/m/..hidden.png', posix)).toBe(true)
  })

  it('compares windows paths case-insensitively and rejects other drives', () => {
    expect(isInside('C:\\Work', 'c:\\work\\a.glb', win32)).toBe(true)
    expect(isInside('C:\\Work', 'D:\\Work\\a.glb', win32)).toBe(false)
  })
})

`src/main/model-url.ts`:
```ts
import type { PlatformPath } from 'node:path'

export const MODEL_SCHEME = 'model'
const LOCAL_HOST = 'file'
const UNC_HOST = 'unc'

const MIME_TYPES: Record<string, string> = {
  '.glb': 'model/gltf-binary',
  '.gltf': 'model/gltf+json',
  '.png': 'image/png',
  '.jpg': 'image/jpeg',
  '.jpeg': 'image/jpeg',
  '.webp': 'image/webp',
  '.ktx2': 'image/ktx2'
}

function isWindows(p: PlatformPath): boolean {
  return p.sep === '\\'
}

export function toModelUrl(filePath: string, p: PlatformPath): string {
  const abs = p.resolve(filePath)
  const host = isWindows(p) && abs.startsWith('\\\\') ? UNC_HOST : LOCAL_HOST
  const segments = abs
    .split(p.sep)
    .filter((s) => s.length > 0)
    .map(encodeURIComponent)
  return `${MODEL_SCHEME}://${host}/${segments.join('/')}`
}

export function fromModelUrl(url: string, p: PlatformPath): string | null {
  let parsed: URL
  try {
    parsed = new URL(url)
  } catch {
    return null
  }
  if (parsed.protocol !== `${MODEL_SCHEME}:`) return null

  let segments: string[]
  try {
    segments = parsed.pathname
      .split('/')
      .filter((s) => s.length > 0)
      .map(decodeURIComponent)
  } catch {
    return null
  }
  if (segments.length === 0) return null

  if (parsed.host === UNC_HOST) {
    return isWindows(p) ? p.normalize(`\\\\${segments.join('\\')}`) : null
  }
  if (parsed.host !== LOCAL_HOST) return null
  return isWindows(p) ? p.normalize(segments.join('\\')) : p.normalize(`/${segments.join('/')}`)
}

export function isInside(dir: string, filePath: string, p: PlatformPath): boolean {
  const rel = p.relative(dir, filePath)
  if (rel === '' || p.isAbsolute(rel)) return false
  return rel.split(p.sep)[0] !== '..'
}

export function mimeType(filePath: string, p: PlatformPath): string {
  return MIME_TYPES[p.extname(filePath).toLowerCase()] ?? 'application/octet-stream'
}
```

import type { PlatformPath } from 'node:path'
import { isInside } from './model-url'

// App-wide, not per window: protocol.handle cannot tell which window sent a request.
export class ModelRegistry {
  private readonly dirs = new Set<string>()
  private missing: string[] = []

  constructor(private readonly p: PlatformPath) {}

  allowFile(filePath: string): string {
    const dir = this.p.dirname(this.p.resolve(filePath))
    this.dirs.add(dir)
    return dir
  }

  isAllowed(filePath: string): boolean {
    for (const dir of this.dirs) {
      if (isInside(dir, filePath, this.p)) return true
    }
    return false
  }

  recordMissing(entry: string): void {
    if (!this.missing.includes(entry)) this.missing.push(entry)
  }

  takeMissing(): string[] {
    const taken = this.missing
    this.missing = []
    return taken
  }
}
```

import type { Session } from 'electron'

const NETWORK_PROTOCOLS = new Set(['http:', 'https:', 'ws:', 'wss:'])

export function shouldBlock(url: string, devServerUrl?: string): boolean {
  let parsed: URL
  try {
    parsed = new URL(url)
  } catch {
    return true
  }
  if (!NETWORK_PROTOCOLS.has(parsed.protocol)) return false
  if (devServerUrl && parsed.host === new URL(devServerUrl).host) return false
  return true
}

// The app runs on an internal network only, so any outbound request is a bug.
export function blockExternalRequests(session: Session, devServerUrl?: string): void {
  session.webRequest.onBeforeRequest((details, callback) => {
    if (shouldBlock(details.url, devServerUrl)) {
      console.warn(`[offline] blocked ${details.url}`)
      callback({ cancel: true })
      return
    }
    callback({})
  })
}
```

import { protocol } from 'electron'
import { readFile } from 'node:fs/promises'
import path from 'node:path'
import type { ModelRegistry } from './model-registry'
import { fromModelUrl, MODEL_SCHEME, mimeType } from './model-url'

/** Must run before app 'ready'. */
export function registerModelScheme(): void {
  protocol.registerSchemesAsPrivileged([
    {
      scheme: MODEL_SCHEME,
      privileges: { standard: true, secure: true, supportFetchAPI: true, corsEnabled: true, stream: true }
    }
  ])
}

export function handleModelProtocol(registry: ModelRegistry): void {
  protocol.handle(MODEL_SCHEME, async (request) => {
    const filePath = fromModelUrl(request.url, path)
    if (!filePath || !registry.isAllowed(filePath)) {
      registry.recordMissing(filePath ?? request.url)
      return new Response('Forbidden', { status: 403 })
    }
    try {
      const data = await readFile(filePath)
      return new Response(data, {
        headers: {
          'content-type': mimeType(filePath, path),
          'access-control-allow-origin': '*'
        }
      })
    } catch {
      registry.recordMissing(filePath)
      return new Response('Not Found', { status: 404 })
    }
  })
}

import { useEffect, type RefObject } from 'react'
import type { ModelViewerElement } from '@google/model-viewer'
import { resetCamera } from '../camera'
import { MESSAGES } from '../messages'

interface ViewerProps {
  src: string
  elementRef: RefObject<ModelViewerElement | null>
  onLoad(): void
  onError(message: string): void
}

export function Viewer({ src, elementRef, onLoad, onError }: ViewerProps) {
  useEffect(() => {
    const el = elementRef.current
    if (!el) return
    const handleLoad = (): void => {
      // A new src keeps the previous camera, so every load starts from the default view.
      resetCamera(el)
      onLoad()
    }
    const handleError = (event: Event): void => {
      const source = (event as CustomEvent<{ sourceError?: unknown }>).detail?.sourceError
      onError(source instanceof Error ? source.message : String(source ?? MESSAGES.unknownError))
    }
    el.addEventListener('load', handleLoad)
    el.addEventListener('error', handleError)
    return () => {
      el.removeEventListener('load', handleLoad)
      el.removeEventListener('error', handleError)
    }
  }, [elementRef, onLoad, onError])

  // Rendering settings are fixed to the modelviewer.dev/editor defaults (see spec §4).
  return (
    <model-viewer
      ref={elementRef}
      src={src}
      camera-controls=""
      environment-image="neutral"
      exposure="1"
      shadow-intensity="1"
      shadow-softness="1"
      tone-mapping="neutral"
      className="viewer"
    />
  )
}

`src/renderer/src/App.tsx`:
```tsx
import { useCallback, useEffect, useRef, useState } from 'react'
import type { ModelViewerElement } from '@google/model-viewer'
import type { OpenedModel, OpenResult } from '../../shared/api'
import { resetCamera } from './camera'
import { EmptyState } from './components/EmptyState'
import { Notice, type NoticeData } from './components/Notice'
import { Toolbar } from './components/Toolbar'
import { Viewer } from './components/Viewer'
import { MESSAGES } from './messages'

export function App() {
  const [model, setModel] = useState<OpenedModel | null>(null)
  const [notice, setNotice] = useState<NoticeData | null>(null)
  const viewerRef = useRef<ModelViewerElement | null>(null)

  const applyResult = useCallback((result: OpenResult | null) => {
    if (!result) return
    if (!result.ok) {
      setNotice({ kind: 'error', text: MESSAGES.unsupported })
      return
    }
    setNotice(null)
    setModel(result.model)
    document.title = result.model.name
  }, [])

  const openDialog = useCallback(async () => {
    applyResult(await window.api.showOpenDialog())
  }, [applyResult])

  const reset = useCallback(() => {
    if (viewerRef.current) resetCamera(viewerRef.current)
  }, [])

  const handleLoad = useCallback(async () => {
    const missing = await window.api.takeMissingFiles()
    setNotice(missing.length > 0 ? { kind: 'warning', text: MESSAGES.missingFiles(missing) } : null)
  }, [])

  const handleError = useCallback(
    async (message: string) => {
      const missing = await window.api.takeMissingFiles()
      const lines = [MESSAGES.loadFailed(model?.name ?? ''), message]
      if (missing.length > 0) lines.push(MESSAGES.missingFiles(missing))
      setNotice({ kind: 'error', text: lines.join('\n') })
    },
    [model]
  )

  useEffect(() => {
    const offOpen = window.api.onMenuOpenResult(applyResult)
    const offReset = window.api.onMenuReset(reset)
    return () => {
      offOpen()
      offReset()
    }
  }, [applyResult, reset])

  useEffect(() => {
    const handleDragOver = (event: DragEvent): void => event.preventDefault()
    const handleDrop = (event: DragEvent): void => {
      event.preventDefault()
      const file = event.dataTransfer?.files[0]
      if (!file) return
      const filePath = window.api.getPathForFile(file)
      if (!filePath) {
        setNotice({ kind: 'error', text: MESSAGES.unsupported })
        return
      }
      void window.api.openPath(filePath).then(applyResult)
    }
    window.addEventListener('dragover', handleDragOver)
    window.addEventListener('drop', handleDrop)
    return () => {
      window.removeEventListener('dragover', handleDragOver)
      window.removeEventListener('drop', handleDrop)
    }
  }, [applyResult])

  return (
    <div className="app">
      <Toolbar
        fileName={model?.name ?? null}
        filePath={model?.path ?? null}
        onOpen={() => void openDialog()}
        onReset={reset}
      />
      <main className="stage">
        {model ? (
          <Viewer src={model.url} elementRef={viewerRef} onLoad={handleLoad} onError={handleError} />
        ) : (
          <EmptyState onOpen={() => void openDialog()} />
        )}
        {notice && <Notice notice={notice} onDismiss={() => setNotice(null)} />}
      </main>
    </div>
  )
}
```
